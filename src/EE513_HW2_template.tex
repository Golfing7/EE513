\section*{Problem 2.1}
Let $X$ be a Gaussian random variable with $m = 5$ and $\sigma^2 = 16$.

\begin{enumerate}[label=(\alph*)]
    \item Using tables of the Q-function, find $P[X > 4]$, $P[X \geq 7]$, $P[2 < X < 7]$, $P[6 \leq X \leq 8]$.
    \begin{solution}
        Given that $m = 5$ and $\sigma^2 = 16$, we can find the $Z$ score values with the formula
        \[
            Z = \frac{x - m}{\sigma} = \frac{x - 5}{4}
        \]
        \begin{align*}
            P[X > 4] &= Q(-0.25) = 1 - Q(0.25) = 0.5987 \\[1em]
            P[X \geq 7] &= Q(0.5) = 0.3085 \\[1em]
            P[2 < X < 7] &= P[X > 2] - P[X > 7] \\
            &= Q(-0.75) - Q(0.5) \\
            &= 1 - 0.2266 - 0.3085 \\
            &= 0.4649 \\[1em]
            P[6 \leq X \leq 8] &= P[X > 6] - P[X > 8] \\
            &= Q(0.25) - Q(0.75) \\
            &= 0.4013 - 0.2266 \\
            &= 0.1747
        \end{align*}
        \qed
    \end{solution}

    \item Find $a$, if $P[X < a] = 0.8869$.
    \begin{solution}
        Given that
        \[
            P[X < a] = 1 - P[X > a] = 0.8869
        \]
        we relate it to
        \[
            Z = \frac{a - m}{\sigma} = \frac{a - 5}{4}
        \]
        By checking the Q-table, $Z = 1.21$.
        \begin{align*}
            1.21 &= \frac{a - 5}{4}\\
            4.84 &= a - 5\\
            a &= 9.84
        \end{align*}
        \qed
    \end{solution}

    \vspace{3em}

    \item Find $c$, if $P[13 < X \leq c] = 0.0123$.
    \begin{solution}
        To break down the probability here, we can do:
        \begin{align*}
            P[13 < X \leq c] &= P[X > 13] - P[X > c]
        \end{align*}
        Where $P[X > 13] = Q(2) = 0.02275$, solving for $P[X > c]$
        \begin{align*}
            0.02275 - P[X > c] = 0.0123 \\
            P[X > c] = 0.01045
        \end{align*}
        Using the same structure as above to solve for $Z$
        \begin{align*}
            Z = 2.31 \\
            Z = \frac{c - 5}{4} \\
            9.24 = c - 5 \\
            14.24 = c
        \end{align*}
        \qed
    \end{solution}

    \vspace{3em}

    \item Show that the Q-function satisfies $Q(-x) = 1 - Q(x)$.
    \begin{solution}
        Definition of $Q(x)$
        \begin{align*}
            Q(X) = \frac{1}{\sqrt{2\pi}} \int_{x}^{\infty} e^{\dfrac{-t^2}{2}} dt \\[1em]
        \end{align*}
        With u substitution
        \begin{align*}
            u &= -t \\
            du &= -dt \\
            Q(-X) &= \frac{1}{\sqrt{2\pi}} \int_{-X}^{\infty} e^{\dfrac{-t^2}{2}} dt \\
            Q(-X) &= \frac{1}{\sqrt{2\pi}} -\int_{X}^{-\infty} e^{\dfrac{-u^2}{2}} du \\
            Q(-X) &= \frac{1}{\sqrt{2\pi}} \int_{-\infty}^{X} e^{\dfrac{-u^2}{2}} du
        \end{align*}
        This last definition happens to be our cdf of a Gaussian random variable
        \begin{align*}
            Q(-X) = F(X)
        \end{align*}
        $Q(X)$ relates to $F(X)$ by $Q(X) = 1 - F(X)$ so
        \begin{align*}
            Q(X) = 1 - Q(-X) \\
            Q(-X) = 1 - Q(X)
        \end{align*}
        \qed
    \end{solution}

    \vspace{3em}

\end{enumerate}

\section*{Problem 2.3}
Let $Y = e^X$, where $X$ is a random variable.

\begin{enumerate}[label=(\alph*)]
    \item Determine the cumulative distribution function and probability density function of $Y$.
    \begin{solution}

        By definition, the cdf of $Y$ is $F_Y(y) = P(Y \leq y) = P(e^X \leq y)$
        We can simplify this by taking the natural log of each side.
        \begin{align*}
            F_Y(y) &= P(Y \leq y) \\
            &= P(e^X \leq y) \\
            &= P(\ln{e^X} \leq \ln{y}) \\
            &= P(X \leq \ln{y}) \\
            &= F_X(\ln{y})
        \end{align*}
        This is assuming strictly for $y > 0$.
        For $y \leq 0$, $F_Y(y) = 0$.

        To obtain the pdf of $Y$, we can take the derivative of $F_Y(y)$ with respect to $y$.
        Strictly for $y > 0$.
        \begin{align*}
            \dfrac{d}{dy} F_Y(y) &= \dfrac{d}{dy} F_X(\ln{y}) \\
            &= f_X(\ln{y})\cdot\dfrac{1}{y}
        \end{align*}

        Therefore,
        \begin{gather*}
            F_Y(y) =
            \begin{cases}
                F_X(\ln{y}), & y > 0 \\
                0, & y \leq 0
            \end{cases}\\
            f_Y(y) =
            \begin{cases}
                f_X(\ln{y})\cdot\dfrac{1}{y}, & y > 0 \\
                0, & y \leq 0
            \end{cases}\\
        \end{gather*}
    \end{solution}

    \vspace{5em}

    \item Determine the probability density function of $Y$ if $X$ is a Gaussian random variable $N(m, \sigma^2)$ with mean $m$ and variance $\sigma^2$:
    \[
        p_X(x) = \frac{1}{\sqrt{2\pi\sigma^2}} e^{-(x-m)^2/(2\sigma^2)}.
    \]
    \begin{solution}

        That is to use this $p_X$ in place of $f_X(\ln{y})$ we found for $f_Y$ in part a.
        For $y \leq 0$, $f_Y(y) = 0$
        \begin{align*}
            f_Y(y) &= f_X(\ln{y}) \cdot \dfrac{1}{y} \\
            &= \left(\frac{1}{\sqrt{2\pi\sigma^2}} e^{-(\ln{y}-m)^2/(2\sigma^2)}\right) \cdot \dfrac{1}{y}
        \end{align*}
        Therefore
        \[
            f_Y(y) =
            \begin{cases}
                \left(\frac{1}{\sqrt{2\pi\sigma^2}} e^{-(\ln{y}-m)^2/(2\sigma^2)}\right) \cdot \dfrac{1}{y}, & y > 0 \\
                0, & y \leq 0
            \end{cases}
        \]
    \end{solution}

    \vspace{5em}

\end{enumerate}

\section*{Problem 2.4}
Let the radius $X$ have the following probability density function (pdf):
\[
    f_X(x) =
    \begin{cases}
        cx(1-x), & 0 < x < 1 \\
        0, & \text{elsewhere}
    \end{cases}
\]

\begin{enumerate}[label=(\alph*)]
    \item Find $c$.
    \begin{solution}

        Given that the cdf of a random variable $X$ approaches 1 as $x \to 1$ (since $f_X$ is only defined on the range $(0, 1)$), we can use that to solve for the problem here.
        Setting up the integral:
        \begin{align*}
            F_X(x) &= P(X \leq x) \\
            &= \int_{0}^{x} (ct - ct^2)dt \\
            &= \int_{0}^{x} ct dt - \int_{0}^{x} ct^2 dt \\
            &= \left[\dfrac{ct^2}{2}\right]_{0}^{x} - \left[\dfrac{ct^3}{3}\right]_{0}^{x} \\
            &= \dfrac{cx^2}{2} - \dfrac{cx^3}{3}
        \end{align*}
        Now that we have the cdf, we can let $x \to 1$ to find $c$.
        \begin{align*}
            1 &= \lim_{x \to 1} F_X(x) \\
            &= \dfrac{c}{2} - \dfrac{c}{3} \\
            &= \dfrac{c}{6} \\
            c &= 6
        \end{align*}
    \end{solution}

    \vspace{3em}

    \item Find the pdf of the area covered by a disk with radius $X$.
    \begin{solution}

        To define this pdf, I'll start with the cdf.
        Keep in mind that since $X$ is the radius, we assume that $X \geq 0$ naturally.
        We also assume that $a \geq 0$ since an area cannot be negative.
        \begin{align*}
            F_A(a) &= P(A \leq a) \\
            &= P(\pi X^2 \leq a) \\
            &= P\left(X \leq \sqrt{\dfrac{a}{\pi}}\right) \\
            F_X\left(\sqrt{\dfrac{a}{\pi}}\right) &= F_A(a)
        \end{align*}
        We want to find the pdf, so we take the derivative.
        \begin{align*}
            \dfrac{d}{da} F_A(a) &= \dfrac{d}{da} F_X\left(\sqrt{\dfrac{a}{\pi}}\right) \\
            f_A(a) &= f_X\left(\sqrt{\dfrac{a}{\pi}}\right)\cdot\left(\dfrac{1}{2\sqrt{a\pi}}\right)
        \end{align*}

        Given that $0 < x < 1$, $0 < a < \pi$.
        We also know $f_X$, so plugging it in we get
        \begin{align*}
            f_A(a) &= f_X\left(\sqrt{\dfrac{a}{\pi}}\right)\cdot\left(\dfrac{1}{2\sqrt{a\pi}}\right) \\
            &= 6\sqrt{\dfrac{a}{\pi}}\left( 1 - \sqrt{\dfrac{a}{\pi}} \right) \cdot \left(\dfrac{1}{2\sqrt{a\pi}}\right) \\
            &= \left(6\sqrt{\dfrac{a}{\pi}} - 6\dfrac{a}{\pi}\right) \cdot \left(\dfrac{1}{2\sqrt{a\pi}}\right) \\
            &= \dfrac{3}{\pi} - \dfrac{3\sqrt{a}}{\pi^{3/2}}
        \end{align*}
    \end{solution}

    \vspace{4em}

    \item Find the pdf of the volume of a sphere with radius $X$.
    \begin{solution}

        Using the volume formula: $V = \dfrac{4}{3}\pi X^3$.
        \begin{align*}
            F_V(v) &= P(V \leq v) \\
            &= P(\dfrac{4}{3} \pi X^3 \leq v) \\
            &= P(X^3 \leq \dfrac{3v}{4\pi}) \\
            &= P(X \leq \sqrt[3]{\dfrac{3v}{4\pi}}) \\
            F_V(v) &= F_X\left(\sqrt[3]{\dfrac{3v}{4\pi}}\right)
        \end{align*}
        \begin{align*}
            0 < x &< 1 \\
            0 < v &< \dfrac{4}{3}\pi \\
            \dfrac{d}{dv} F_V(v) &= \dfrac{d}{dv} F_X\left(\sqrt[3]{\dfrac{3v}{4\pi}}\right) \\
            &= f_X\left(\sqrt[3]{\dfrac{3v}{4\pi}}\right) \cdot \left(\dfrac{\sqrt[3]{3}}{3\sqrt[3]{4\pi}v^{2/3}}\right) \\
            &= 6\left(\sqrt[3]{\dfrac{3v}{4\pi}} - \left(\dfrac{3v}{4\pi}\right)^{2/3}\right) \cdot \left(\dfrac{\sqrt[3]{3}}{3\sqrt[3]{4\pi}v^{2/3}}\right) \\
            &= 6\left( \dfrac{1}{3^{1/3} (4\pi)^{2/3} v^{1/3} } - \dfrac{1}{4\pi} \right)
        \end{align*}
        The domain of this being $0 < v < \dfrac{4}{3}\pi$ with $f_V(v) = 0$ outside of that.
    \end{solution}

    \vspace{4em}

    \item Find the pdf of $Y = X^n$.
    \begin{solution}

        The domain is $0 < x < 1$ as previously defined for X and $0 < y < 1$.
        \begin{align*}
            F_Y(y) &= P(Y \leq y) \\
            &= P(X^n \leq y) \\
            &= P(X \leq \sqrt[n]{y}) \\
            F_Y(y) &= F_X(\sqrt[n]{y})
        \end{align*}
        \begin{align*}
            \dfrac{d}{dy} F_Y(y) &= \dfrac{d}{dy} F_X(\sqrt[n]{y}) \\
            &= f_X(\sqrt[n]{y})\cdot \left( \dfrac{\sqrt[n]{y}}{yn} \right) \\
            &= 6\sqrt[n]{y}\left(1 - \sqrt[n]{y}\right) \cdot \left( \dfrac{\sqrt[n]{y}}{yn} \right) \\
            &= \left(6\sqrt[n]{y} - 6y^{2/n}\right) \cdot \left( \dfrac{\sqrt[n]{y}}{yn} \right) \\
            &= 6\dfrac{y^{2/n}}{yn} - 6 \dfrac{y^{3/n}}{yn}
        \end{align*}
        Outside $0 < y < 1$, $f_Y(y) = 0$
    \end{solution}

    \vspace{4em}

\end{enumerate}

\section*{Problem 2.5}
This problem examines the relationship between $E[X]$ and $E\!\left[\frac{1}{X}\right]$.

\begin{enumerate}[label=(\alph*)]
    \item Suppose that
    \[
        X =
        \begin{cases}
            -1, & \text{with probability } \frac{1}{9} \\
            \frac{1}{2}, & \text{with probability } \frac{4}{9} \\
            2, & \text{with probability } \frac{4}{9}
        \end{cases}
    \]
    Show that $E\!\left[\dfrac{1}{X}\right] = \dfrac{1}{E[X]}$.
    \begin{solution}

        We can check both to see if this is true.
        \begin{align*}
            E\!\left[\dfrac{1}{X}\right] &= -1\cdot\dfrac{1}{9} + 2\cdot\dfrac{4}{9} + \dfrac{1}{2}\cdot\dfrac{4}{9} \\
            &= -\dfrac{1}{9} + \dfrac{8}{9} + \dfrac{2}{9} \\
            &= 1
        \end{align*}
        For the second part
        \begin{align*}
            E[X] &= -1\cdot\dfrac{1}{9} + \dfrac{1}{2}\cdot\dfrac{4}{9} + 2\cdot\dfrac{4}{9} \\
            &= -\dfrac{1}{9} + \dfrac{2}{9} + \dfrac{8}{9} \\
            &= 1 \\
            \dfrac{1}{E[X]} &= 1
        \end{align*}
        So $E\!\left[\dfrac{1}{X}\right] = \dfrac{1}{E[X]}$
    \end{solution}

    \vspace{5em}

    \item Suppose that $X$ is uniformly distributed on the interval $(1,2)$. Show that $E(1/X) \neq 1/E(X)$.
    \begin{solution}

        Given that $X$ is uniformly distributed over $(1, 2)$
        \[
            f_X(x) =
            \begin{cases}
                1, & 1 < x < 2 \\
                0, & \text{elsewhere}
            \end{cases}
        \]
        The cdf for $x \in (1, 2)$
        \begin{align*}
            F_X(x) &= \int_{1}^{x} dx \\
            &= \left[x\right]_{1}^{x} \\
            &= x - 1
        \end{align*}
        \[
            F_X(x) =
            \begin{cases}
                0, & x \leq 1 \\
                x - 1, & 1 < x < 2 \\
                1, & x \geq 2
            \end{cases}
        \]

        Now that we have $F_X(x)$, we can show what was originally asked.
        \begin{align*}
            E\left[\dfrac{1}{x}\right] &= \int_{1}^{2} \dfrac{1}{x} dx \\
            &= \left[\ln{x}\right]_{1}^{2} \\
            &= \ln{2}
        \end{align*}
        For the other one
        \begin{align*}
            E[x] &= \int_{1}^{2} x dx \\
            &= \left[\dfrac{x^2}{2}\right]_{1}^{2} \\
            &= 2 - \dfrac{1}{2} \\
            &= \dfrac{3}{2} \\
            \dfrac{1}{E[X]} &= \dfrac{2}{3} \\
            \dfrac{2}{3} &\neq \ln{2}
        \end{align*}
        \qed
    \end{solution}

    \vspace{5em}

\end{enumerate}

\section*{Problem 2.6}
Let $X$ be a Laplacian-distributed random variable; that is,
\[
    f_X(x) = \frac{\alpha}{2} \exp(-\alpha |x|), \qquad -\infty < x < +\infty.
\]

\begin{enumerate}[label=(\alph*)]
    \item Determine the characteristic function $M_X(jv) = E[e^{jvX}]$ of $X$.
    \begin{solution}
        \begin{align*}
            M_X(jv) = E[e^{jvX}] &= \int_{-\infty}^{\infty} \exp(jvx) \cdot \frac{\alpha}{2} \exp(-\alpha |x|) dx \\
            &\text{Splitting the integral} \\
            &= \int_{-\infty}^{0} \exp(jvx) \cdot \frac{\alpha}{2} \exp(\alpha x) dx + \int_{0}^{\infty} \exp(jvx) \cdot \frac{\alpha}{2} \exp(-\alpha x) dx \\
            &= \frac{\alpha}{2} \int_{-\infty}^{0} \exp(x(jv + \alpha)) dx + \frac{\alpha}{2} \int_{0}^{\infty} \exp(x(jv - \alpha)) dx \\
            &= \frac{\alpha}{2} \left[\dfrac{\exp(x(jv + \alpha))}{jv + \alpha}\right]_{-\infty}^{0} + \frac{\alpha}{2} \left[\dfrac{\exp(x(jv - \alpha))}{jv - \alpha}\right]_{0}^{\infty} \\
            &= \frac{\alpha}{2jv + 2\alpha} + \frac{\alpha}{2} \left( \frac{1}{\alpha - jv} \right) \\
            &= \frac{\alpha}{2jv + 2\alpha} + \frac{\alpha}{2\alpha - 2jv} \\
            &= \frac{\alpha\cdot(2\alpha - 2jv)}{(2jv + 2\alpha)\cdot(2\alpha - 2jv)} + \frac{\alpha\cdot(2jv + 2\alpha)}{(2\alpha - 2jv)\cdot(2jv + 2\alpha)} \\
            &= \frac{\alpha\cdot(2\alpha - 2jv) + \alpha\cdot(2jv + 2\alpha)}{(2jv + 2\alpha)\cdot(2\alpha - 2jv)} \\
            &= \frac{2\alpha^2 - 2jv\alpha + 2jv\alpha + 2\alpha^2}{(2jv + 2\alpha)\cdot(2\alpha - 2jv)} \\
            &= \frac{4\alpha^2}{(2jv + 2\alpha)\cdot(2\alpha - 2jv)} \\
            &= \frac{4\alpha^2}{4jv\alpha - 4j^2 v^2 + 4\alpha^2 - 4jv\alpha} \\
            &= \frac{\alpha^2}{v^2 + \alpha^2}
        \end{align*}
    \end{solution}

    \vspace{5em}

    \item Determine the mean and variance of $X$ by applying the moment property of the characteristic function.
    \begin{solution}

        To calculate the mean, we need to find the first derivative of the moment function.
        \begin{align*}
            M'_X(jv) &= \dfrac{d}{dv} M_X(jv) \\
            &= \dfrac{d}{dv} (\alpha^2)(v^2 + \alpha^2)^{-1} \\
            &= -2\alpha^2 v(v^2 + \alpha^2)^{-2}
        \end{align*}

        At $v = 0$,
        \begin{align*}
            M'_X(0) &= 0
        \end{align*}

        To find $E[X^2]$, we take the derivative of this again.
        \begin{align*}
            M''_X(jv) &= \dfrac{d}{dv} M'_X(jv) \\
            &= -\alpha^2 \dfrac{d}{dv} (v^2 + \alpha^2)^{-2}\cdot(2v) \\
            &= -\alpha^2\cdot(-2\cdot(v^2 + \alpha^2)^{-3}\cdot 4v^2 + (v^2 + \alpha^2)^{-2} \cdot 2) \\
            &= 8\alpha^2 v^2(v^2 + \alpha^2)^{-3} - 2\alpha^2(v^2 + \alpha^2)^{-2}
        \end{align*}

        At $v = 0$,
        \begin{align*}
            M''_X(0) &= 8\alpha^2 (0)((0) + \alpha^2)^{-3} - 2\alpha^2((0) + \alpha^2)^{-2} \\
            &= - 2\alpha^2(\alpha^2)^{-2} \\
            &= - \dfrac{2}{\alpha^2}
        \end{align*}

        \begin{align*}
            E[X] &= \dfrac{1}{j} M'_X(jv) |_{v = 0} = 0 \\
            E[X^2] &= \dfrac{1}{j^2} M''_X(jv) |_{v = 0} = \dfrac{2}{\alpha^2} \\
            Var(X) &= E[X^2] - (E[X])^2 = \dfrac{2}{\alpha^2}
        \end{align*}

    \end{solution}

    \vspace{5em}

\end{enumerate}

\section*{Problem 2.7}
The sample $X$ of a speech signal is a Laplacian random variable with parameter $\alpha = 1$. Suppose $X$ is quantized by a nonuniform quantizer consisting of four intervals:
$(-\infty, -a]$, $(-a, 0]$, $(0, a]$, and $(a, +\infty)$.

\begin{enumerate}[label=(\alph*)]
    \item Find the value of $a$ so that $X$ is equally likely to fall in each of the four intervals.
    \begin{solution}

        We were given the formula for a Laplace distribution in the above question.
        \[
            f_X(x) = \frac{\alpha}{2} \exp(-\alpha |x|), \qquad -\infty < x < +\infty.
        \]
        Given that $\alpha = 1$, we simplify this to
        \[
            f_X(x) = \frac{1}{2} \exp(-|x|), \qquad -\infty < x < +\infty.
        \]
        We effectively want to find $a$ such that
        \[
            \int_{-\infty}^{-a} f_X(x) dx = \int_{-a}^{0} f_X(x) dx = \int_{0}^{a} f_X(x) dx = \int_{a}^{\infty} f_X(x) dx
        \]
        Given that this is over the entire line and $\int_{-\infty}^{\infty} f_X(x) = 1$ by definition.
        The above four integrals are over the entire space, so if they are to be equal, they must all be equal to $\dfrac{1}{4}$.
        Because of this, we can find a single integral where $a$ makes it equal to $\dfrac{1}{4}$.
        \begin{align*}
            \dfrac{1}{4} &= \int_{a}^{\infty} f_X(x) dx \\
            &= \int_{a}^{\infty} \frac{1}{2} \exp(-x) dx \qquad \text{$|x| = x, a > 0$} \\
            &= \left[-\frac{1}{2} \exp(-x)\right]_{a}^{\infty} \\
            &= \frac{1}{2} \exp(-a) \\
            \vspace{4em} \\
            \exp(-a) &= \dfrac{1}{2} \\
            -a &= \ln{\dfrac{1}{2}} \\
            a &= \ln{2}
        \end{align*}
    \end{solution}

    \vspace{4em}

    \item Find the representation point $x_1 = q(X)$ for $X$ in $(0,a]$ that minimizes the mean-square error,
    \[
        \int_0^a (x - x_1)^2 f_X(x)\, dx.
    \]
    \textit{Hint: Differentiate with respect to $x_1$.} Find representation points for the other intervals.

    \begin{solution}

        \begin{align*}
            \int_0^a \dfrac{\delta}{\delta x_1}(x - x_1)^2 f_X(x)\, dx &= \int_0^a (-2x + 2x_1) f_X(x)\, dx \\
            &= \int_0^a (-2x + 2x_1) \cdot (\frac{1}{2} \exp(-|x|))\, dx \\
            &= \int_0^a -x \exp(-x) + x_1 \exp(-x)\, dx \qquad \text{$|x| = x, a > 0$} \\
            &= \left[ x \exp(-x) + \exp(-x) \right]_{0}^{a} + \left[ - x_1 \exp(-x) \right]_{0}^{a} \\
            &= a \exp(-a) + \exp(-a) - 1 - x_1 \exp(-a) + x_1
        \end{align*}

        Setting this equal to $0$ in order to find representation points and plugging in $a = \ln{2}$
        \begin{align*}
            0 &= \ln{2} \exp(-\ln{2}) + \exp(-\ln{2}) - 1 - x_1 \exp(-\ln{2}) + x_1 \\
            &= \dfrac{\ln{2}}{2} + \dfrac{1}{2} - 1 - \dfrac{x_1}{2} + x_1 \\
            &= \dfrac{\ln{2}}{2} - \dfrac{1}{2} + \dfrac{x_1}{2} \\
            \dfrac{1}{2} - \dfrac{\ln{2}}{2} &= \dfrac{x_1}{2} \\
            x_1 &= 1 - \ln{2}
        \end{align*}

        Next, find it in the interval $(a, \infty)$
        \begin{align*}
            &\cdots \\
            &\left[ x \exp(-x) + \exp(-x) \right]_{a}^{\infty} + \left[ - x_1 \exp(-x) \right]_{a}^{\infty} \\
            &= - a\exp(-a) - \exp(-a) + x_1 \exp(-a)
        \end{align*}

        Now substituting in $a = \ln{2}$ when equal to $0$.
        \begin{align*}
            0 &= - a\exp(-a) - \exp(-a) + x_1 \exp(-a) \\
            x_1 \exp(-a) &= a\exp(-a) + \exp(-a) \\
            \dfrac{x_1}{2} &= \dfrac{\ln{2}}{2} + \dfrac{1}{2} \\
            x_1 &= \ln{2} + 1
        \end{align*}

        And given the symmetry of this equation around $x = 0$. \\
        For $(-\infty, -a]$, $x_1 = - \ln{2} - 1$ \\
        For $(-a, 0]$, $x_1 = \ln{2} - 1$ \\
        For $[0, a)$, $x_1 = 1 - \ln{2}$ \\
        For $[a, \infty)$, $x_1 = \ln{2} + 1$

    \end{solution}
    \vspace{5em}

    \item Evaluate the mean-square error of the quantizer: $E[(X - q(X))^2]$.
    \begin{solution}

        Setting up the equations, because of the even symmetry, again given that $a = \ln{2}$
        \begin{align*}
            E[(X - q(X))^2] &= 2\left(\int_0^a (x - 1 + \ln{2})^2 f_X(x)\, dx + \int_a^\infty (x - \ln{2} - 1)^2 f_X(x)\, dx\right)
        \end{align*}

        Working out the first integral, plugging back in $x_1 = 1 - \ln{2}$ for simplicity.
        \begin{align*}
            \int_0^a (x - x_1)^2 f_X(x)\, dx &= \dfrac{1}{2} \int_0^a (x - x_1)^2 \exp(-x)\, dx \\
            &= \dfrac{1}{2} \left( \int_0^a x^2 \exp(-x)\, dx - 2x_1 \int_0^a x \exp(-x)\, dx + x_1^2 \int_0^a \exp(-x)\, dx \right)
        \end{align*}

        Solving each of these separately, using $\exp(-\ln{2}) = \dfrac{1}{2}$
        \begin{align*}
            \int_0^a x^2 \exp(-x)\, dx &= \left[ -x^2 \exp(-x) - 2x \exp(-x) - 2\exp(-x) \right]_{0}^{a} \\
            &= -\dfrac{(\ln{2})^2}{2} - \ln{2} - 1 + 2 \\
            &= 1 - \ln{2} - \dfrac{(\ln{2})^2}{2} \\[1em]
            \int_0^a x \exp(-x)\, dx &= \left[ -x \exp(-x) - \exp(-x) \right]_{0}^{a} \\
            &= -\dfrac{\ln{2}}{2} - \dfrac{1}{2} + 1 \\
            &= \dfrac{1 - \ln{2}}{2} = \dfrac{x_1}{2} \\[1em]
            \int_0^a \exp(-x)\, dx &= \left[ -\exp(-x) \right]_{0}^{a} \\
            &= -\dfrac{1}{2} + 1 = \dfrac{1}{2}
        \end{align*}

        Plugging these back in
        \begin{align*}
            \int_0^a (x - x_1)^2 f_X(x)\, dx &= \dfrac{1}{2} \left( 1 - \ln{2} - \dfrac{(\ln{2})^2}{2} - 2x_1 \cdot \dfrac{x_1}{2} + \dfrac{x_1^2}{2} \right) \\
            &= \dfrac{1}{2} \left( 1 - \ln{2} - \dfrac{(\ln{2})^2}{2} - \dfrac{x_1^2}{2} \right) \\
            &= \dfrac{1}{2} \left( 1 - \ln{2} - \dfrac{(\ln{2})^2}{2} - \dfrac{1 - 2\ln{2} + (\ln{2})^2}{2} \right) \\
            &= \dfrac{1}{2} \left( \dfrac{1}{2} - (\ln{2})^2 \right) \\
            &= \dfrac{1}{4} - \dfrac{(\ln{2})^2}{2}
        \end{align*}

        Working out the second integral, plugging back in $x_2 = \ln{2} + 1$ for simplicity.
        \begin{align*}
            \int_a^\infty (x - x_2)^2 f_X(x)\, dx &= \dfrac{1}{2} \int_a^\infty (x - x_2)^2 \exp(-x)\, dx \\
            &= \dfrac{1}{2} \left( \int_a^\infty x^2 \exp(-x)\, dx - 2x_2 \int_a^\infty x \exp(-x)\, dx + x_2^2 \int_a^\infty \exp(-x)\, dx \right)
        \end{align*}

        Solving each of these separately, again using $\exp(-\ln{2}) = \dfrac{1}{2}$
        \begin{align*}
            \int_a^\infty x^2 \exp(-x)\, dx &= \left[ -x^2 \exp(-x) - 2x \exp(-x) - 2\exp(-x) \right]_{a}^{\infty} \\
            &= 0 + \dfrac{(\ln{2})^2}{2} + \ln{2} + 1 \\
            &= 1 + \ln{2} + \dfrac{(\ln{2})^2}{2} \\[1em]
            \int_a^\infty x \exp(-x)\, dx &= \left[ -x \exp(-x) - \exp(-x) \right]_{a}^{\infty} \\
            &= 0 + \dfrac{\ln{2}}{2} + \dfrac{1}{2} \\
            &= \dfrac{\ln{2} + 1}{2} = \dfrac{x_2}{2} \\[1em]
            \int_a^\infty \exp(-x)\, dx &= \left[ -\exp(-x) \right]_{a}^{\infty} \\
            &= 0 + \dfrac{1}{2} = \dfrac{1}{2}
        \end{align*}

        Plugging these back in
        \begin{align*}
            \int_a^\infty (x - x_2)^2 f_X(x)\, dx &= \dfrac{1}{2} \left( 1 + \ln{2} + \dfrac{(\ln{2})^2}{2} - 2x_2 \cdot \dfrac{x_2}{2} + \dfrac{x_2^2}{2} \right) \\
            &= \dfrac{1}{2} \left( 1 + \ln{2} + \dfrac{(\ln{2})^2}{2} - \dfrac{x_2^2}{2} \right) \\
            &= \dfrac{1}{2} \left( 1 + \ln{2} + \dfrac{(\ln{2})^2}{2} - \dfrac{1 + 2\ln{2} + (\ln{2})^2}{2} \right) \\
            &= \dfrac{1}{2} \left( \dfrac{1}{2} \right) \\
            &= \dfrac{1}{4}
        \end{align*}

        Putting both integrals back into the original equation
        \begin{align*}
            E[(X - q(X))^2] &= 2\left( \dfrac{1}{4} - \dfrac{(\ln{2})^2}{2} + \dfrac{1}{4} \right) \\
            &= 2\left( \dfrac{1}{2} - \dfrac{(\ln{2})^2}{2} \right) \\
            &= 1 - (\ln{2})^2 \\
            &\approx 0.5195
        \end{align*}
        \qed
    \end{solution}

\end{enumerate}
