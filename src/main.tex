    \item Construct an appropriate probability model $(S, F, P)$ with sixteen sample points $\xi$, each representing a state of the network. Specify $S$, $F$, and $P$.

    \item Let $A = \{\xi : \text{node-1 and node-4 can communicate}\}$. Calculate $P(A)$.

    \begin{solution}

        This is all $\xi$ such that a path exists between node-1 and node-4.
        Examining the graph, it appears that there are two paths it can take. $a \rightarrow c \rightarrow b$ OR $a \rightarrow d$.
        That is $B = \{(1, 1, 1, x): x \in \{0, 1\}\}$ and $C = \{(1, x, x, 1): x \in \{0, 1\}\}$.
        Therefore, $P(B \cup C) = P(B) + P(C) - P(B \cap C)$ \\
        Individually,
        \[P(B) = P(\{1, 1, 1, 0\}) + P(\{1, 1, 1, 1\}) = p^{3}(1-p) + p^4\] \\
        \[P(C) = P(\{1, 0, 0, 1\}) + P(\{1, 0, 1, 1\}) + P(\{1, 1, 0, 1\}) + P(\{1, 1, 1, 1\}) = p^2(1 - p)^2 + 2(p^3(1 - p)) + p^4\] \\

        \begin{align}
            P(B \cup C) &= p^{3}(1-p) + p^4 + p^2(1 - p)^2 + 2(p^3(1 - p)) + p^4 - p^4\\
            &= p^2 + p^3 - p^4

    \item Show that
    \[
        P(A) = p\, P(A \mid \text{link-c available}) + (1-p)\, P(A \mid \text{link-c not available}).
    \]
    Use this formula to recalculate $P(A)$ by inspection.

    \begin{solution}

        Instead of finding individual probabilities, we can use the definition of conditional probability.
        Keep in mind that, by definition, $P(\text{link-c available}) = p$ and $P(\text{link-c not available}) = 1 - p$.
        \[p\, P(A \mid \text{link-c available}) = p\, \frac{P(A \cap \text{link-c available})}{p} = P(A \cap \text{link-c available})\]
        Likewise,
        \[(1-p)\, P(A \mid \text{link-c not available}) = (1-p)\, \frac{P(A \cap \text{link-c not available})}{1-p} = P(A \cap \text{link-c not available})\]
        Now we can simplify,
        \[P(A) = P(A \cap \text{link-c available}) + P(A \cap \text{link-c not available})\]
        Given $A \cap \text{link-c available}$ and $A \cap \text{link-c not available}$ and the events being mutually exclusive,
        \[(A \cap \text{link-c available}) \cup (A \cap \text{link-c not available}) = A\]
        Therefore,
        \[P(A) = p\, P(A \mid \text{link-c available}) + (1-p)\, P(A \mid \text{link-c not available})\] \\
        To confirm that these are equal by inspection, first lets look at $P(A \mid \text{link-c available})$
        Given that link c is available in the network,
        \[P(A \mid \text{link-c available}) = 2p^2(1-p) + p^3\] \\
        \[P(A \mid \text{link-c not available}) = p^2(1-p) + p^3\]
        By our found defintion,
        \begin{align}
            P(A) &= p(2p^2(1-p) + p^3) + (1 - p)(p^2(1-p) + p^3) \\
            &= 2p^3(1-p) + p^4 + p^2(1-p)^2 + p^3(1-p) \\
            &= -p^4 + p^3 + p^2
        \end{align}
        Which is the same as our original finding of A.

    \end{solution}

    \item Prove that $P(A)$ would be increased by moving link-c between node-1 and node-2 rather than between node-2 and node-3.

        Which is true for all $p < 1$.

    \end{solution}

    \section*{Problem 1.2}

    Let $A_1, A_2, \ldots, A_N$ be $N$ events defined on some probability space. Prove that
    \[
        P\left(\bigcup_{n=1}^{N} A_n\right) = \sum_{1 \le i \le N} P(A_i) - \sum_{1 \le i < j \le N} P(A_i \cap A_j) + \sum_{1 \le i < j < k \le N} P(A_i \cap A_j \cap A_k) - \cdots \pm P(A_1 \cap A_2 \cap \cdots \cap A_N).
    \]
    \emph{Hint: Prove the statement by induction.}

    \begin{solution}

        First lets begin with a base case of 1.
        \[P\left(\bigcup_{n=1}^{1} A_n\right) = P(A_1)\]
        Now a base case of 2.
        \[P\left(\bigcup_{n=1}^{2} A_n\right) = P(A_1) + P(A_2) - P(A_1 \cap A_2)\]
        This can be proven like so.
        \[A_1 \cup A_2 = (A_1 \cap A_2') \cup (A_1 \cap A_2) \cup (A_1' \cap A_2)\]
        Since these are all mutually exclusive, we can take the probability like this:
        \[P(A_1 \cup A_2) = P(A_1 \cap A_2') + P(A_1 \cap A_2) + P(A_1' \cap A_2)\] \\
        We can define each individually probability like this:
        \begin{align}
            P(A_1) &= P(A_1 \cap A_2) + P(A_1 \cap A_2') \\
            P(A_2) &= P(A_1 \cap A_2) + P(A_1' \cap A_2) \\
            P(A_1) - P(A_1 \cap A_2) &= P(A_1 \cap A_2') \\
            P(A_2) - P(A_1 \cap A_2) &= P(A_1' \cap A_2)
        \end{align}
        Plugging them back into the original equation:
        \begin{align}
            P(A_1 \cup A_2) &= P(A_1 \cap A_2') + P(A_1 \cap A_2) + P(A_1' \cap A_2) \\
            P(A_1 \cup A_2) &= P(A_1) - P(A_1 \cap A_2) + P(A_1 \cap A_2) + P(A_2) - P(A_1 \cap A_2) \\
            P(A_1 \cup A_2) &= P(A_1) + P(A_2) - P(A_1 \cap A_2)
        \end{align}
        Thus, we've proven the base case of N=2.
        Now, we assume that it works for a case of N=m.
        \[
            P\left(\bigcup_{n=1}^{m} A_n\right) = \sum_{1 \le i \le m} P(A_i) - \sum_{1 \le i < j \le m} P(A_i \cap A_j) + \cdots \pm P(A_1 \cap A_2 \cap \cdots \cap A_m).
        \]
        And attempt to prove that it works for N=m + 1.
        Let $B = \bigcup_{n=1}^{m} A_n$, then by our inductive hypothesis:
        \begin{align}
            P(B \cup A_{m + 1}) &= P(B) + P(A_{m + 1}) - P(B \cap A_{m + 1}) \\
            P\left(\bigcup_{n=1}^{m+1} A_n\right) &= P(B) + P(A_{m + 1}) - P(B \cap A_{m + 1}) \\
            &= P\left(\bigcup_{n=1}^{m} A_n\right) + P(A_{m + 1}) - P\left(\left(\bigcup_{n=1}^{m} A_n\right) \cap A_{m + 1}\right) \\
            &= P\left(\bigcup_{n=1}^{m} A_n\right) + P(A_{m + 1}) - P\left(\bigcup_{n=1}^{m} \left(A_n \cap A_{m + 1}\right)\right) \text{By distribution} \\
            &\text{Then apply inductive hypothesis again to $\bigcup_{n=1}^{m} \left(A_n \cap A_{m + 1}\right)$} \\
            &= P\left(\bigcup_{n=1}^{m} A_n\right) + P(A_{m + 1}) - (\sum_{1 \le i \le m} P(A_i \cap A_{m + 1})) + \cdots \pm P(A_1 \cap A_2 \cap \cdots \cap A_m \cap A_{m+1}) \\
            &\text{Then by combining terms:} \\
            &= \sum_{1 \le i \le m+1} P(A_i) - \sum_{1 \le i < j \le m+1} P(A_i \cap A_j) + \cdots \pm P(A_1 \cap A_2 \cap \cdots \cap A_m \cap A_{m+1})
        \end{align}
        \qed
    \end{solution}

    \section*{Problem 1.3}

    A random experiment $A$ has $N$ mutually exclusive outcomes $A_1, A_2, \ldots, A_N$, and random experiment $B$ has $M$ mutually exclusive outcomes $B_1, B_2, \ldots, B_M$. Establish Bayes' rule,
    \[
        \Pr(B_m \mid A_n) = \frac{\Pr(A_n \mid B_m)\Pr(B_m)}{\sum_{i=1}^{M} \Pr(A_n \mid B_i)\Pr(B_i)}.
    \]

    \begin{solution}

        Let's begin with definitions, for any two events $A, B$
        \begin{align}
            \Pr(A \mid B) &= \frac{\Pr(A \cap B)}{\Pr(B)} \\
            \Pr(A \mid B)\Pr(B) &= \Pr(A \cap B) \\
            \Pr(B \mid A) &= \frac{\Pr(A \cap B)}{\Pr(A)} \\
            \Pr(B \mid A) &= \frac{\Pr(A \mid B)}{\Pr(B)\Pr(A)}
        \end{align}
        With this, we establish a relation between $\Pr(A \mid B)$, $\Pr(A \cap B)$, and $\Pr(B \mid A)$. \\
        Now, using the law of total probability (since $B$ is a partition and $A_i$ is mutually exclusive),
        \begin{align}
            \Pr(A_n) &= \sum_{i=1}^{M}\Pr(A_n \mid B_i)\Pr(B_i)
        \end{align}
        Using the original definition of conditional probability
        \begin{align}
            \Pr(B_m \mid A_n) &= \frac{\Pr(B_m \cap A_n)}{\Pr(A_n)} \\
            \Pr(B_m \mid A_n) &= \frac{\Pr(B_m \cap A_n)}{\sum_{i=1}^{M}\Pr(A_n \mid B_i)\Pr(B_i)} \\
            \Pr(B_m \mid A_n) &= \frac{\Pr(A_n \mid B_m) \Pr(B_m)}{\sum_{i=1}^{M}\Pr(A_n \mid B_i)\Pr(B_i)}
        \end{align}
        \qed

    \end{solution}

    \begin{solution}

        \[
            x = X,\qquad y = Y - X, \qquad z = 1 - Y
        \]
        And our inequalities
        \begin{align}
            X &< \frac{1}{2} \\
            Y - X &< \frac{1}{2} \\
            1 - Y &< \frac{1}{2}

    \section*{Problem 1.5}
    \emph{(Papoulis, based on Problem 4-21, p. 85)}

    The probability of heads of a random coin is a RV $\phi$ uniform in the interval $(0,1)$.

    \begin{enumerate}[label=(\alph*)]

    \item Find $P\{0.3 \le \phi \le 0.7\}$.

    \begin{solution}

        Since P is uniform, $\Pr(\phi \le a) = a$
        \[
            \Pr(\phi \le 0.7) - \Pr(\phi \le 0.3) = 0.7 - 0.3 = 0.4
        \]
        \qed

    \end{solution}

    \item The coin is tossed 10 times and heads shows 6 times. Find the a posteriori probability that $\phi$ is between 0.3 and 0.7.

    \begin{solution}

        That is, to find $\Pr(0.3 \le \phi \le 0.7 \mid \text{6 heads in 10 tosses})$.
        We need to find everything for the following equation
        \[
            f_{\phi \mid data}(x) = \frac{f_{\phi}(x) \cdot \Pr(\text{6 heads in 10} \mid \phi = x)}{\Pr(\text{6 heads})}
        \]
        \begin{align}
            f_{\phi}(x) &= 1 \\
            f_{\phi}(x)\Pr(\text{6 heads in 10} \mid \phi = x) = \binom{10}{6}\phi^6(1-\phi)^4
        \end{align}
        We need to integrate this over the interval 0--1 using the Beta function
        \begin{align}
            \binom{10}{6} \int_{0}^{1} x^6(1-x)^4 dx &= \binom{10}{6} \cdot B(7, 5) \\
        \end{align}
        Wit the beta function being
        \begin{align}
            B(7, 5) &= \frac{6!4!}{11!} \\
        \end{align}
        Plugging this back in
        \begin{align}
            \binom{10}{6} \cdot B(7, 5) &= \binom{10}{6} \cdot \frac{6!4!}{11!} \\
            &= \frac{10!}{6!4!} \cdot \frac{6!4!}{11!} \\
            &= \frac{10!}{11!} \\
            &= \frac{1}{11} \\
            &= \Pr(\text{6 heads})
        \end{align}
        Going back to our original equation
        \begin{align}
            f_{\phi \mid data} &= \frac{f_{\phi}(x) \cdot \Pr(\text{6 heads in 10} \mid \phi = x)}{\Pr(\text{6 heads})} \\
            f_{\phi \mid data} &= 11 \cdot \Pr(\text{6 heads in 10} \mid \phi = x) \\
            \Pr(0.3 \leq \phi \leq 0.7) &= \int_{0.3}^{0.7} f_{\phi \mid data}(x)dx = 11 \binom{10}{6} \int_{0.3}^{0.7} x^6(1-x)^{4}dx \\
        \end{align}

        \begin{align}
            \Pr(0.3 \le \phi \le 0.7 \mid \text{data}) &= 11\binom{10}{6}\int_{0.3}^{0.7} x^6(1-x)^4\,dx
        \end{align}

        Expanding $(1-x)^4$ via the binomial theorem:
        \begin{align}
            (1-x)^4 &= \sum_{j=0}^{4} \binom{4}{j}(-x)^j = \binom{4}{0} - \binom{4}{1}x + \binom{4}{2}x^2 - \binom{4}{3}x^3 + \binom{4}{4}x^4
        \end{align}

        Multiplying through by $x^6$:
        \begin{align}
            x^6(1-x)^4 &= \binom{4}{0}x^6 - \binom{4}{1}x^7 + \binom{4}{2}x^8 - \binom{4}{3}x^9 + \binom{4}{4}x^{10}
        \end{align}

        Integrating term by term:
        \begin{align}
            \int_{0.3}^{0.7} x^6(1-x)^4\,dx
            &= \left[\binom{4}{0}\frac{x^7}{7} - \binom{4}{1}\frac{x^8}{8} + \binom{4}{2}\frac{x^9}{9} - \binom{4}{3}\frac{x^{10}}{10} + \binom{4}{4}\frac{x^{11}}{11}\right]_{0.3}^{0.7}
        \end{align}

        Evaluating at the bounds:
        \begin{align}
            &= \left(\frac{(0.7)^7}{7} - \frac{4(0.7)^8}{8} + \frac{6(0.7)^9}{9} - \frac{4(0.7)^{10}}{10} + \frac{(0.7)^{11}}{11}\right) \notag\\
            &\quad - \left(\frac{(0.3)^7}{7} - \frac{4(0.3)^8}{8} + \frac{6(0.3)^9}{9} - \frac{4(0.3)^{10}}{10} + \frac{(0.3)^{11}}{11}\right)
        \end{align}

        \begin{align}
            \int_{0.3}^{0.7} x^6(1-x)^4\,dx &\approx 0.0003325005331\dots
        \end{align}

        Therefore:
        \begin{align}
            \Pr(0.3 \le \phi \le 0.7 \mid \text{data}) &\approx 11\binom{10}{6}\cdot0.0003325005331 \approx 0.7680762316
        \end{align}
        \qed

    \end{solution}

    \end{enumerate}

    \section*{Problem 1.6}
    \emph{(based on Problem 2.97 on page 91, Leon-Garcia'06)}

    A block of 50 bits is transmitted over a binary communication channel with probability of bit error $p = 10^{-2}$.

    \begin{enumerate}[label=(\alph*)]

    \item The continuous random variable $X$ has a \emph{Laplacian} probability density function
    \[
        f_X(x) = a e^{-b|x|}, \qquad -\infty < x < +\infty,
    \]

    \begin{solution}

        To find the relationship, we can integrate over the entire domain to ensure that it's equal to 1.
        \begin{align}
            \int_{-\infty}^{\infty} a e^{-b|x|} dx &= 2a\int_{0}^{\infty} e^{-bx} dx \\
            &= 2a \left[-\frac{e^{-bx}}{b}\right]_{0}^{\infty} \\
            &= 2a \left(\frac{1}{b}\right)
        \end{align}
        Using the fact that this must equal 1,
        \begin{align}
            2a \left(\frac{1}{b}\right) &= 1 \\
            2a &= b
        \end{align}

        Now given $a = \frac{1}{2}$, $b = 1$ and our original \emph{PDF}
        \begin{align}
            f_X(x) = \frac{1}{2} e^{-|x|}
        \end{align}

        And to determine the \emph{CDF}, we handle for $x \le 0$ and $x \geq 0$ separately. \\
        For $x \geq 0$:
        \begin{align}
            F_X(x) &= \int_{-\infty}^{0} \frac{1}{2} e^{-t} dt + \int_{0}^{x} \frac{1}{2} e^{-t} dt
        \end{align}

        Evaluating the first of the two integrals:
        \begin{align}
            \int_{-\infty}^{0} \frac{1}{2} e^{-|t|} dt &= \frac{1}{2} \left[e^{t}\right]_{-\infty}^{0} \\
            &= \frac{1}{2} (1)
            &= \frac{1}{2}
        \end{align}

        The second of the two give us:
        \begin{align}
            \int_{0}^{x} \frac{1}{2} e^{-t} dt &= \frac{1}{2} \left[-e^{-t}\right]_{0}^{x}\\
            &= \frac{1}{2} \left(-e^{-x} + 1\right) \\
            &= \frac{-e^{-x}}{2} + \frac{1}{2}

        \]


    \end{solution}
